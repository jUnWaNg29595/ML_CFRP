===========================================
 常用Git命令
===========================================

git status                    # 查看当前状态
git add <文件>                # 添加文件到暂存区
git commit -m "说明"          # 提交暂存区
git push origin main          # 推送到远程
git pull origin main          # 拉取远程更新
git log --oneline             # 查看提交历史
git diff                      # 查看修改内容
git checkout -- <文件>        # 撤销修改
git rm --cached <文件>        # 移除跟踪但不删除文件
git clone <地址>              # 克隆仓库
